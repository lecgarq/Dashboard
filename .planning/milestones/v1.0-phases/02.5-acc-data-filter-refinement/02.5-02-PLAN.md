const DEFAULT_FILTERS: GraphFilters = {

  companyRoles: [],
  dateFrom: "",
  dateTo: "",

<!-- Module logic inversion + new dimensions -->
```typescript
// Module: exclude only when ALL user's modules are off; users with no modules always pass.
if (filters.disabledModules.length > 0 && node.modules.length > 0) {
  if (node.modules.every((m) => filters.disabledModules.includes(m))) return false;
}

// companyRole: null -> "Unspecified" bucket
if (filters.companyRoles.length > 0) {
  const bucket = node.companyRole ?? "Unspecified";
  if (!filters.companyRoles.includes(bucket)) return false;
}

// Date range (inclusive both ends, ISO YYYY-MM-DD lexicographic compare)
if (filters.dateFrom || filters.dateTo) {
  if (!node.lastSignIn) return false;  // exclude unknown when range active
  const d = node.lastSignIn.slice(0, 10);
  if (filters.dateFrom && d < filters.dateFrom) return false;
  if (filters.dateTo   && d > filters.dateTo)   return false;
}
```
