4. **Alpha mask via color multiplication (Pitfall 6 — MeshBasicMaterial.opacity is global; per-instance alpha must be baked into color):**
   ```typescript
   const DIM = 0.15;
   function applyAlphaMask3D(mask: Float32Array): void {
     for (let i = 0; i < n; i++) {
       const lit = mask[i] >= 0.99 ? 1.0 : DIM;
       instanceColor.setXYZ(
         i,
         props.nodeColors[i*4]   * lit,
         props.nodeColors[i*4+1] * lit,
         props.nodeColors[i*4+2] * lit,
       );
     }
     instanceColor.needsUpdate = true;
   }
   ```

5. **Position pump:**
   ```typescript
   function pumpPositions3D(xyz: Float32Array): void {
     for (let i = 0; i < n; i++) {
       dummy.position.set(xyz[i*3], xyz[i*3+1], xyz[i*3+2]);
       dummy.updateMatrix();
       mesh.setMatrixAt(i, dummy.matrix);
     }
     mesh.instanceMatrix.needsUpdate = true;
   }
   ```

6. **Self-driving render rAF (separate from useGraphRafLoop):** OrbitControls damping requires per-frame `controls.update()` regardless of whether new positions arrived. Internal rAF that calls `controls.update(); renderer.render(scene, camera);` every frame while mounted. Cancel on cleanup.

7. **fitView():**
   ```typescript
   function fitView(): void {
     const box = new THREE.Box3();
     const tmpVec = new THREE.Vector3();
     const tmpMat = new THREE.Matrix4();
     for (let i = 0; i < n; i++) {
       mesh.getMatrixAt(i, tmpMat);
       tmpVec.setFromMatrixPosition(tmpMat);
       box.expandByPoint(tmpVec);
     }
     const size = box.getSize(new THREE.Vector3());
     const center = box.getCenter(new THREE.Vector3());
     const maxDim = Math.max(size.x, size.y, size.z) || 1;
     const fov = camera.fov * (Math.PI / 180);
     const distance = (maxDim / (2 * Math.tan(fov / 2))) * 1.5;
     camera.position.set(center.x, center.y, center.z + distance);
     controls.target.copy(center);
     controls.update();
   }
   ```

8. **Resize handling:** `ResizeObserver` on containerRef.current. On resize → setSize / update camera.aspect / camera.updateProjectionMatrix() / fitView().

9. **Cleanup:** cancel rAF, remove canvas, dispose renderer + geometry + material + ResizeObserver.

     // fitView first so the bounding-box-derived "end" pose is correct,
     // then tween from a top-down-ish start (high y, small z offset) to the fitted pose.
     if (prevMode.current === '2d' && props.mode === '3d') {
       handle3D.current?.fitView();
       const endPos = cam.position.clone();
       const startPos = endPos.clone();
       startPos.y += Math.max(Math.abs(endPos.z), 1) * 0.75; // tilt-down origin
       const DURATION = 600; // ms — LOCKED
       const t0 = performance.now();
       const easeInOutCubic = (t: number) =>
         t &lt; 0.5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2;
       const step = (now: number) => {
         const p = Math.min(1, (now - t0) / DURATION);
         const e = easeInOutCubic(p);
         cam.position.lerpVectors(startPos, endPos, e);
         if (p &lt; 1) {
           animFrame.current = requestAnimationFrame(step);
         } else {
           animFrame.current = null;
         }
       };
       animFrame.current = requestAnimationFrame(step);
     }

     // 3D → 2D: 400ms z → 0 flatten (CONTEXT.md locked).
     // Tween camera.position.z toward 0 over 400ms.
     if (prevMode.current === '3d' && props.mode === '2d') {
       const startZ = cam.position.z;
       const DURATION = 400; // ms — LOCKED
       const t0 = performance.now();
       const easeOutCubic = (t: number) => 1 - Math.pow(1 - t, 3);
       const step = (now: number) => {
         const p = Math.min(1, (now - t0) / DURATION);
         const e = easeOutCubic(p);
         cam.position.z = startZ * (1 - e); // → 0
         if (p &lt; 1) {
           animFrame.current = requestAnimationFrame(step);
         } else {
           animFrame.current = null;
         }
       };
       animFrame.current = requestAnimationFrame(step);
     }

     prevMode.current = props.mode;
     return () => {
       if (animFrame.current != null) {
         cancelAnimationFrame(animFrame.current);
         animFrame.current = null;
       }
     };
   }, [props.mode]);
