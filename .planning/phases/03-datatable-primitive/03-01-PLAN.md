DataTable public API (from @tanstack/react-table types):
```typescript
import type { ColumnDef, Row, SortingState } from "@tanstack/react-table";

export interface DataTableProps<T> {
  data: T[];                                   // pre-filtered/prepared by the caller
  columns: ColumnDef<T>[];                     // typed to T
  onRowClick?: (row: Row<T>) => void;          // fires on non-chevron click
  renderExpanded?: (row: Row<T>) => React.ReactNode; // inline peek slot
  pinnedColumn?: string;                       // column id to pin left (e.g. "name")
  defaultSort?: SortingState;                  // applied via initialState (resets each visit)
  hasActiveFilter?: boolean;                   // selects which empty-state message shows
  onClearFilters?: () => void;                 // wired by the empty-state "Clear filters" action
  emptyMessage?: string;                       // override "no data yet"
  filteredEmptyMessage?: string;               // override "no results — adjust filters"
  className?: string;
}
export function DataTable<T>(props: DataTableProps<T>): React.ReactElement;
```

Density persistence key (locked): localStorage key `datatable-density`, values `"comfortable" | "compact"`.
