    **usePredicateEngine.ts** — implement exactly the predicate from RESEARCH Pattern 1:
    ```typescript
    export function usePredicateEngine(inputs: PredicateInputs): void {
      useEffect(() => {
        inputs.physics.setMask((i: number): number => {
          const f = inputs.features[i];
          if (inputs.isolatedNodeIndex !== null) return i === inputs.isolatedNodeIndex ? 1.0 : 0.15;
          for (const [dim, allowed] of Object.entries(inputs.activeFilters)) {
            if (allowed.size === 0) continue;
            const v = featureValueForDim(f, dim);  // small helper: maps dimId → bucket/value
            if (!allowed.has(v)) return 0.15;
          }
          if (inputs.searchQuery && !f.nameLower.startsWith(inputs.searchQuery) && !f.emailLower.startsWith(inputs.searchQuery)) return 0.15;
          if (inputs.lassoSelection) {
            if (!inputs.lassoSelection.has(i)) return 0.15;
            if (inputs.drillDown) {
              for (const [dim, value] of Object.entries(inputs.drillDown)) {
                if (featureValueForDim(f, dim) !== value) return 0.15;
              }
            }
          }
          return 1.0;
        });
      }, [inputs.physics, inputs.features, inputs.activeFilters, inputs.searchQuery, inputs.lassoSelection, inputs.drillDown, inputs.isolatedNodeIndex]);
    }
    ```
