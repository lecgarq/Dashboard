Persistence rules:
- `AccFolder.upsert({ where:{id: folder.id}, ... })` using URN as PK.
- `AccFolderPermission.upsert({ where:{ folderId_roleId: {...} } })` — schema has `@@unique([folderId, roleId])`.
- For each permission, compute `permType` via `mapActions(perm.actions).tier ?? "View Only"` (null tier defensively floors to "View Only"; the `extended`/`extendedActions` info is reconstructable at render time from raw `actions`, so we don't need new columns).
- `AccProject.update({ where:{id: project.id}, data:{ folderCrawlStatus: result.status } })` after each project completes.
- Soft-delete: folders not seen in this crawl pass should NOT be hard-deleted (matches Phase 2 MemberAggregator pattern for `accMemberCache` — additive only). Stale rows tolerable; missing rows would zero the matrix.

```javascript
#!/usr/bin/env node
require('tsx/cjs');
const { PrismaClient } = require('@prisma/client');
const { extractAndPersistFolders } = require('../lib/acc/folderCrawl.ts');
const { get2LeggedAutodeskToken } = require('../lib/server/acc-admin.ts');
const pLimit = require('p-limit').default;

(async () => {
  const prisma = new PrismaClient();
  try {
    const token = await get2LeggedAutodeskToken();
    const projects = await prisma.accProject.findMany({ where: { status: 'active' }, select: { id: true, accountId: true, name: true, apsHubId: true } });
    const hubId = projects[0]?.apsHubId; // or however hubId is sourced — match Plan 02's script
    const limit = pLimit(5);
    await Promise.all(projects.map(p => limit(() => extractAndPersistFolders(prisma, hubId, p, token).catch(err => {
      console.error(`[folder-crawl-cron] ${p.name} failed`, err);
      return { folderCount: 0, permissionCount: 0, status: 'failed' };
    }))));
    console.log('[folder-crawl-cron] complete');
  } finally {
    await prisma.$disconnect();
  }
})();
```
