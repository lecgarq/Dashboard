    ```javascript
    #!/usr/bin/env node
    require('tsx/cjs');
    const path = require('node:path');
    const fs = require('node:fs');
    const { PrismaClient } = require('@prisma/client');
    const { runDcIngest, isKillSwitchActive } = require('../lib/acc/dcIngest');

    const REPO_ROOT = path.join(__dirname, '..');

    async function main() {
      if (isKillSwitchActive(REPO_ROOT)) {
        console.log('[dc-daily-ingest] Kill switch active (.dc-ingest.disabled) — exiting.');
        process.exit(0);
      }
      const prisma = new PrismaClient();
      try {
        const result = await runDcIngest(prisma);
        console.log(`[dc-daily-ingest] Run ${result.ingestRunId} ended with status=${result.status}`);
        console.log(`[dc-daily-ingest] projectsProcessed=${result.projectsProcessed} quotaUsed=${result.quotaUsed}`);
        if (result.errorMessage) console.error(`[dc-daily-ingest] ERROR: ${result.errorMessage}`);
        process.exit(result.status === 'success' || result.status === 'killed' ? 0 : 1);
      } finally {
        await prisma.$disconnect();
      }
    }

    main().catch((e) => { console.error('[dc-daily-ingest] Fatal:', e); process.exit(1); });
    ```

    **scripts/dc-daily-cron.ps1** — extend the existing harness with kill-switch + 30-minute timeout + logging:

    ```powershell
    # scripts/dc-daily-cron.ps1
    # Windows Task Scheduler entry point. Schedule daily at 03:00 local Hermosillo time.
    $ErrorActionPreference = 'Stop'
    Set-Location $PSScriptRoot\..

    $logDir = Join-Path $PSScriptRoot '..\logs'
    if (-not (Test-Path $logDir)) { New-Item -ItemType Directory -Path $logDir | Out-Null }
    $logFile = Join-Path $logDir ("dc-ingest-" + (Get-Date -Format 'yyyy-MM-dd') + ".log")

    # Kill switch — second layer in addition to runtime check.
    if (Test-Path '.dc-ingest.disabled') {
      "$(Get-Date -Format 'o') Kill switch present (.dc-ingest.disabled) — skipping run." | Tee-Object -FilePath $logFile -Append
      exit 0
    }

    # Run with 30-minute timeout (CONTEXT.md run-duration budget).
    $proc = Start-Process -FilePath 'node' -ArgumentList 'scripts/dc-daily-ingest.cjs' `
      -RedirectStandardOutput $logFile -RedirectStandardError $logFile -PassThru -NoNewWindow
    if (-not $proc.WaitForExit(30 * 60 * 1000)) {
      "$(Get-Date -Format 'o') TIMEOUT after 30min — killing process." | Tee-Object -FilePath $logFile -Append
      $proc.Kill()
      exit 124
    }
    exit $proc.ExitCode
    ```
