```typescript
import { subDays, max as dateMax, startOfDay } from 'date-fns';

export const PROJECT_BATCH_LIMIT = 50;     // APS hard cap, evidenced in dc-ingest-where-i-admin.cjs:36
export const SLICE_DAYS = 30;              // CONTEXT-locked window size
export const OVERLAP_DAYS = 1;             // CONTEXT-locked 1-day overlap for late-arriving events

export interface ProjectProgress {
  projectId: string;
  earliestCovered: Date | null;
  latestCovered: Date | null;
  projectCreatedAt: Date;
  newProjectFlag: boolean;
}

export interface Slice {
  projectIds: string[];     // up to PROJECT_BATCH_LIMIT
  start: Date;
  end: Date;
  reason: 'backward' | 'forward' | 'new-project';
}

export interface DailyPlan {
  slices: Slice[];
  totalProjects: number;
  estimatedQuota: number;   // = slices.length
}

export function planDailySlice(
  projects: ProjectProgress[],
  yesterdayUtc: Date,             // injectable for tests
): DailyPlan;

export function applySliceCompletion(
  prev: ProjectProgress,
  slice: { start: Date; end: Date; reason: Slice['reason'] },
): ProjectProgress;
```

Key edge cases (CONTEXT-derived):
- New project: emit single slice with `reason: 'new-project'`, projectIds=[id], start=yesterday-30d, end=yesterday. Accelerated-backfill flag triggers a *second* deeper slice on the next run.
- Fully backfilled (earliestCovered <= projectCreatedAt AND latestCovered >= yesterday): emit nothing for that project.
- Backward edge floor: `backwardStart = max(subDays(earliestCovered, 30), projectCreatedAt)`. If equal to earliestCovered, no slice.
- Forward edge: when `latestCovered < yesterday`, emit slice `[latestCovered-1d, yesterday]` (overlap window).
- Batching: group slices with identical (start,end) into one Slice with up to 50 projectIds; spill into a second Slice if a 51st project has the same window.
