export interface BulkAccProject {
  id: string;
  name: string;
  status: string;
  isAdmin: boolean;
  roles: string[];        // role NAME strings (not IDs)
  modules: string[];      // product keys where tier !== "none"
}

export interface BulkAccUser {
  email: string;
  name: string;
  found: boolean;                  // true for any synced user
  projectCount: number;
  activeCount: number;             // count of projects where status === "active"
  adminCount: number;              // count of projects where isAdmin === true
  hasNoProjects: boolean;
  syncedAt: string;                // ISO
  allRoles: string[];              // de-duped union across projects
  allModules: string[];            // de-duped union across projects
  projects: BulkAccProject[];
  companyRole?: string | null;     // null acceptable for v2.0 (HQ v1 enrichment deferred)
  lastSignIn?: string | null;      // pick the MAX across per-project lastSignIn values
  isAccountAdmin: boolean;         // requires hub-level check — see decision below
  addedOn: string | null;          // earliest addedOn across per-project rows
}

2. `export function buildCacheBlob(entry: MemberAggregatorEntry, syncedAt: Date): BulkAccUser`:
   - Build `projects: BulkAccProject[]` by mapping each `entry.perProject` record:
     ```typescript
     {
       id: pp.projectId,
       name: pp.projectName,
       status: "active",                          // v2 always-fresh after sync (deleted projects are filtered out of the aggregator upstream)
       isAdmin: pp.projectAdmin,
       roles: pp.roleNames,                       // raw name strings (matches v1.0 cache shape)
       modules: Object.entries(pp.products).filter(([, tier]) => tier !== "none").map(([key]) => key),
     }
     ```
   - Compute derived fields:
     - `projectCount = projects.length`
     - `activeCount = projects.filter(p => p.status === "active").length`
     - `adminCount = projects.filter(p => p.isAdmin).length`
     - `hasNoProjects = projects.length === 0`
     - `allRoles = Array.from(new Set(projects.flatMap(p => p.roles)))`
     - `allModules = Array.from(new Set(projects.flatMap(p => p.modules)))`
     - `lastSignIn = entry.perProject.map(p => p.lastSignIn).filter(Boolean).sort().reverse()[0] ?? null` (most recent ISO across projects)
     - `addedOn = entry.perProject.map(p => p.addedOn).filter(Boolean).sort()[0] ?? null` (earliest ISO)
   - Return full `BulkAccUser` typed object with `companyRole: null`, `isAccountAdmin: false`, `found: true`, `email: entry.email`, `name: entry.name`, `syncedAt: syncedAt.toISOString()`.
   - The return-type annotation `: BulkAccUser` is mandatory — TypeScript will reject any missing/wrong-typed field at compile time. This is the structural safety net for Pitfall 2.

```javascript
async function runQuickSyncShell() {
  const startedAt = new Date();
  console.log("[release] Quick Sync: invoking lib/acc/quick-sync-extraction.ts via tsx...");

  const result = spawnSync("npx", ["tsx", "lib/acc/quick-sync-extraction.ts"], {
    stdio: "inherit",
    timeout: TIMEOUT_MS - 30_000,   // leave 30s buffer before the outer watchdog kills the process
    shell: process.platform === "win32",
    env: { ...process.env },
  });

  if (result.error || result.status !== 0) {
    const reason = result.error
      ? `tsx invocation failed: ${result.error.message}`
      : `Quick Sync extraction exited ${result.status}`;
    throw new Error(reason);
  }

  // Success: mark SyncMeta('quick','success'). The TS extractor handles its own logging
  // of project/member counts; release.cjs only owns the SyncMeta status row.
  const prisma = createPrisma();
  try {
    await prisma.syncMeta.upsert({
      where: { id: "quick" },
      create: { id: "quick", lastRunAt: startedAt, lastStatus: "success", lastError: null },
      update: { lastRunAt: startedAt, lastStatus: "success", lastError: null },
    });
    console.log("[release] Quick Sync complete; SyncMeta('quick','success') written.");
  } finally {
    await prisma.$disconnect().catch(() => {});
  }
}
```
