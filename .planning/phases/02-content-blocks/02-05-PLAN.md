In the upload handler, replace or augment the `fetch('/api/wiki-media', ...)` call to use `XMLHttpRequest` for progress tracking:

```typescript
// Replace the fetch call with XHR to get upload progress
function uploadWithProgress(formData: FormData, fileName: string): Promise<{ url: string }> {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open('POST', '/api/wiki-media');

    xhr.upload.addEventListener('progress', (e) => {
      if (e.lengthComputable) {
        setUploadProgress({ fileName, progress: Math.round((e.loaded / e.total) * 100) });
      }
    });

    xhr.addEventListener('load', () => {
      setUploadProgress(null);
      if (xhr.status >= 200 && xhr.status < 300) {
        resolve(JSON.parse(xhr.responseText));
      } else {
        reject(new Error(`Upload failed: ${xhr.status}`));
      }
    });

    xhr.addEventListener('error', () => {
      setUploadProgress(null);
      reject(new Error('Upload network error'));
    });

    xhr.send(formData);
  });
}
```
