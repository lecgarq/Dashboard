Create `components/clash/wiki-editor/image-node.tsx`. This replaces `tiptap-extension-resize-image` (ImageResize) completely. Do NOT modify the existing VideoNode — this is a separate file following the same pattern.

**Node attributes:**
- `src: string | null` — Google Drive URL or proxy URL
- `alt: string | null` — alt text for accessibility
- `width: number | null` — current display width in px
- `alignment: 'left' | 'center' | 'right' | 'full-width'` — default: 'full-width'
- `caption: string | null` — optional caption text

**ImageNodeView React component** (`"use client"` implicitly via WikiEditor):

State:
- `lightboxOpen: boolean` — controlled by click on image
- `captionValue: string` — local input state, synced to node attrs on blur
- `showAltInput: boolean` — shown via right-click context (use a `contextmenu` event listener)

Alignment CSS mapping:
```typescript
const alignmentClass = {
  'left': 'mr-auto',
  'center': 'mx-auto',
  'right': 'ml-auto',
  'full-width': 'w-full',
};
```

Floating alignment toolbar: Show when `selected === true`. Position using absolute positioning above the image. Four buttons: Left | Center | Right | Full-width. Each calls `updateAttributes({ alignment: value })`. Use existing `Button` component from `@/components/ui/button` with `variant="ghost"` and `size="sm"`.

Resize: Same corner drag handle pattern as VideoNode (mousedown + mousemove + mouseup on document).

Lightbox: When `lightboxOpen` is true, render a fixed full-screen overlay (z-50, bg-black/80) with the image centered and a close button (X icon, top-right). Clicking the overlay or X closes it. Clicking the image itself opens it.

GIF detection: Check if `src` ends in `.gif` or `attrs.src?.includes('image/gif')`. If GIF, render as `<img>` with no special treatment — browser handles autoplay.

Right-click menu (simulated via `onContextMenu`): Show a small dropdown with:
- "Replace media" (calls a passed `onReplace` handler — wired in WikiEditor.tsx via editor commands)
- "Copy link" (calls `navigator.clipboard.writeText(attrs.src ?? '')`)
- "Edit alt text" (shows the `showAltInput` field)

**Node definition:**
```typescript
export const ImageNode = Node.create({
  name: "image",  // Must match "image" to replace the existing ImageResize which parses <img> tags
  group: "block",
  selectable: true,
  draggable: true,
  atom: true,
  addAttributes() {
    return {
      src: { default: null },
      alt: { default: null, parseHTML: el => el.getAttribute('alt'), renderHTML: attrs => attrs.alt ? { alt: attrs.alt } : {} },
      width: { default: null, parseHTML: el => el.style.width ? parseInt(el.style.width) : null, renderHTML: attrs => attrs.width ? { style: `width: ${attrs.width}px` } : {} },
      alignment: { default: 'full-width', parseHTML: el => el.getAttribute('data-alignment') || 'full-width', renderHTML: attrs => ({ 'data-alignment': attrs.alignment }) },
      caption: { default: null, parseHTML: el => el.getAttribute('data-caption'), renderHTML: attrs => attrs.caption ? { 'data-caption': attrs.caption } : {} },
    };
  },
  parseHTML() { return [{ tag: 'img[src]' }]; },
  renderHTML({ HTMLAttributes }) {
    return ['img', mergeAttributes(HTMLAttributes, { class: 'max-w-full rounded-xl' })];
  },
  addNodeView() { return ReactNodeViewRenderer(ImageNodeView); },
});
```

**Important:** The node name MUST be `"image"` so it parses existing `<img>` tags already in the database. This also means WikiEditor.tsx must remove `ImageResize` from the extensions array and add `ImageNode` in its place (done in 02-05).

**Full structure of the ImageNodeView component:**

```typescript
"use client";
import { useState, useCallback } from "react";
import { Node, mergeAttributes } from "@tiptap/core";
import { NodeViewWrapper, ReactNodeViewRenderer, type ReactNodeViewProps } from "@tiptap/react";
import { cn } from "@/lib/core/utils";
import { Button } from "@/components/ui/button";
import { AlignLeft, AlignCenter, AlignRight, Maximize2, X } from "lucide-react";

function ImageNodeView({ node, selected, updateAttributes }: ReactNodeViewProps) {
  const attrs = node.attrs as { src: string | null; alt: string | null; width: number | null; alignment: string; caption: string | null };
  const [lightboxOpen, setLightboxOpen] = useState(false);
  const [captionValue, setCaptionValue] = useState(attrs.caption ?? '');
  const [showContextMenu, setShowContextMenu] = useState(false);
  const [contextMenuPos, setContextMenuPos] = useState({ x: 0, y: 0 });

  const handleDrag = useCallback((event: React.MouseEvent) => {
    event.preventDefault();
    const startX = event.pageX;
    const startWidth = attrs.width || (event.currentTarget.parentElement?.offsetWidth ?? 400);
    const onMouseMove = (e: MouseEvent) => updateAttributes({ width: Math.max(80, startWidth + (e.pageX - startX)) });
    const onMouseUp = () => { document.removeEventListener('mousemove', onMouseMove); document.removeEventListener('mouseup', onMouseUp); };
    document.addEventListener('mousemove', onMouseMove);
    document.addEventListener('mouseup', onMouseUp);
  }, [attrs.width, updateAttributes]);

  const alignmentStyle: Record<string, string> = {
    'left': 'mr-auto', 'center': 'mx-auto', 'right': 'ml-auto', 'full-width': 'w-full',
  };

  return (
    <NodeViewWrapper className="relative my-4">
      <div
        className={cn(
          "group relative inline-block overflow-hidden rounded-xl transition-all",
          alignmentStyle[attrs.alignment] ?? 'w-full',
          selected ? "ring-2 ring-primary" : ""
        )}
        style={{ width: attrs.width ? `${attrs.width}px` : attrs.alignment === 'full-width' ? '100%' : undefined }}
      >
        {/* Alignment toolbar — shown when selected */}
        {selected && (
          <div className="absolute -top-9 left-1/2 z-10 flex -translate-x-1/2 items-center gap-0.5 rounded-lg border border-border bg-background p-0.5 shadow-lg">
            {(['left', 'center', 'right', 'full-width'] as const).map(align => (
              <Button
                key={align}
                variant={attrs.alignment === align ? "secondary" : "ghost"}
                size="sm"
                className="h-7 px-2 text-xs"
                onClick={() => updateAttributes({ alignment: align })}
              >
                {align === 'left' ? <AlignLeft size={12} /> : align === 'center' ? <AlignCenter size={12} /> : align === 'right' ? <AlignRight size={12} /> : '↔'}
              </Button>
            ))}
          </div>
        )}

        {/* Image */}
        <img
          src={attrs.src ?? ''}
          alt={attrs.alt ?? ''}
          className="block h-auto w-full cursor-pointer rounded-xl border border-border/50 shadow-lg"
          onClick={() => setLightboxOpen(true)}
          onContextMenu={e => { e.preventDefault(); setShowContextMenu(true); setContextMenuPos({ x: e.clientX, y: e.clientY }); }}
          draggable={false}
        />

        {/* Resize handle — bottom-right corner */}
        <div
          className="pointer-events-auto absolute bottom-0 right-0 flex h-4 w-4 cursor-nwse-resize items-center justify-center rounded-tl-md bg-primary/80 opacity-0 backdrop-blur-sm group-hover:opacity-100"
          onMouseDown={handleDrag}
        >
          <div className="h-1.5 w-1.5 rounded-full bg-white shadow-sm" />
        </div>
      </div>

      {/* Caption */}
      <input
        type="text"
        placeholder="Add caption..."
        value={captionValue}
        onChange={e => setCaptionValue(e.target.value)}
        onBlur={() => updateAttributes({ caption: captionValue || null })}
        className="mt-1 w-full border-0 bg-transparent text-center text-xs text-muted-foreground placeholder:text-muted-foreground/50 focus:outline-none"
      />

      {/* Right-click context menu */}
      {showContextMenu && (
        <div
          className="fixed z-50 min-w-[140px] rounded-lg border border-border bg-popover p-1 shadow-lg"
          style={{ top: contextMenuPos.y, left: contextMenuPos.x }}
        >
          <button
            className="flex w-full items-center rounded px-2 py-1.5 text-sm hover:bg-accent"
            onClick={() => { navigator.clipboard.writeText(attrs.src ?? ''); setShowContextMenu(false); }}
          >Copy link</button>
          <button
            className="flex w-full items-center rounded px-2 py-1.5 text-sm hover:bg-accent"
            onClick={() => setShowContextMenu(false)}
          >Edit alt text</button>
          <div className="my-1 h-px bg-border" />
          <button
            className="flex w-full items-center rounded px-2 py-1.5 text-sm text-destructive hover:bg-accent"
            onClick={() => setShowContextMenu(false)}
          >Replace media</button>
        </div>
      )}
      {showContextMenu && <div className="fixed inset-0 z-40" onClick={() => setShowContextMenu(false)} />}

      {/* Lightbox */}
      {lightboxOpen && (
        <div
          className="fixed inset-0 z-50 flex items-center justify-center bg-black/80"
          onClick={() => setLightboxOpen(false)}
        >
          <button
            className="absolute right-4 top-4 rounded-full bg-white/10 p-2 text-white hover:bg-white/20"
            onClick={() => setLightboxOpen(false)}
          >
            <X size={20} />
          </button>
          <img
            src={attrs.src ?? ''}
            alt={attrs.alt ?? ''}
            className="max-h-[90vh] max-w-[90vw] rounded-xl object-contain shadow-2xl"
            onClick={e => e.stopPropagation()}
          />
        </div>
      )}
    </NodeViewWrapper>
  );
}
```
