**File 1: `table-utils.ts`**

Create utility functions for table data operations:

```typescript
import type { Editor } from '@tiptap/core';
import * as XLSX from 'xlsx';

// Walk the ProseMirror document tree to extract a 2D array of cell text from the table at cursor position.
// Returns string[][] where [0] is header row (if present).
export function extractTableRows(editor: Editor): string[][] {
  const rows: string[][] = [];
  const { selection, doc } = editor.state;
  // Find the nearest table ancestor of the selection
  let tableNode = null;
  doc.nodesBetween(0, doc.content.size, (node) => {
    if (node.type.name === 'table') tableNode = node;
  });
  if (!tableNode) return rows;
  tableNode.forEach((rowNode: any) => {
    const row: string[] = [];
    rowNode.forEach((cellNode: any) => {
      row.push(cellNode.textContent);
    });
    rows.push(row);
  });
  return rows;
}

// Export table at cursor as .xlsx file. Called from right-click context menu.
export function exportTableAsXlsx(editor: Editor, fileName = 'table-export.xlsx') {
  const rows = extractTableRows(editor);
  if (rows.length === 0) return;
  const ws = XLSX.utils.aoa_to_sheet(rows);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, 'Sheet1');
  XLSX.writeFile(wb, fileName);
}
```

Create a React hook and component for client-side sort and filter (these are VIEW-only — they do not modify the Yjs document, they only control what rows are displayed).

```typescript
"use client";
import { useState, useCallback } from 'react';

export interface SortState {
  columnIndex: number | null;
  direction: 'asc' | 'desc';
}

export interface FilterState {
  query: string;
  active: boolean;
}

// Hook that manages sort and filter state for a table node view.
// Returns sorted/filtered rows derived from the raw rows prop.
export function useSortFilter(rawRows: string[][]) {
  const [sort, setSort] = useState<SortState>({ columnIndex: null, direction: 'asc' });
  const [filter, setFilter] = useState<FilterState>({ query: '', active: false });

  const toggleSort = useCallback((colIndex: number) => {
    setSort(prev =>
      prev.columnIndex === colIndex
        ? { columnIndex: colIndex, direction: prev.direction === 'asc' ? 'desc' : 'asc' }
        : { columnIndex: colIndex, direction: 'asc' }
    );
  }, []);

  // Separate header row (index 0) from data rows (index 1+) before sort/filter
  const headerRow = rawRows[0] ?? [];
  let dataRows = rawRows.slice(1);

  if (filter.active && filter.query.trim()) {
    const q = filter.query.toLowerCase();
    dataRows = dataRows.filter(row => row.some(cell => cell.toLowerCase().includes(q)));
  }

  if (sort.columnIndex !== null) {
    const col = sort.columnIndex;
    dataRows = [...dataRows].sort((a, b) => {
      const cmp = (a[col] ?? '').localeCompare(b[col] ?? '', undefined, { numeric: true });
      return sort.direction === 'asc' ? cmp : -cmp;
    });
  }

  return { sort, filter, toggleSort, setFilter, displayRows: [headerRow, ...dataRows] };
}

// FilterBar component — shown/hidden via toolbar toggle
export function FilterBar({ filter, setFilter }: {
  filter: FilterState;
  setFilter: (f: FilterState) => void;
}) {
  return (
    <div className="flex items-center gap-2 border-b border-border/50 bg-muted/30 px-3 py-1.5">
      <input
        type="text"
        placeholder="Filter rows..."
        value={filter.query}
        onChange={e => setFilter({ ...filter, query: e.target.value })}
        className="h-7 flex-1 rounded border border-border bg-background px-2 text-sm focus:outline-none focus:ring-1 focus:ring-primary"
      />
      <button
        onClick={() => setFilter({ query: '', active: false })}
        className="text-xs text-muted-foreground hover:text-foreground"
      >
        Clear
      </button>
    </div>
  );
}
```
