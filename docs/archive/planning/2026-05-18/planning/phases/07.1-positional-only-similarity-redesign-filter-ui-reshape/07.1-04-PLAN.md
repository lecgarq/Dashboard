Add pure helper:
```typescript
export function buildLinkStrengthBuffer(
  renderLinkCount: number,
  forceStrengths: Float32Array,
  renderLinkDefaultStrength = 0,  // 0 = let cosmos default 1/degree apply (Pitfall 2 mitigation)
): Float32Array {
  const out = new Float32Array(renderLinkCount + forceStrengths.length);
  out.fill(renderLinkDefaultStrength, 0, renderLinkCount);
  out.set(forceStrengths, renderLinkCount);
  return out;
}
```
