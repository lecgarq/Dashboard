```javascript
// scripts/dc-ingest-where-i-admin.cjs:128-147
async function listProjectAdminProjects(accountId, token2Leg) {
  // NOTE: the filter[accessLevel]=projectAdmin server-side filter is misleading —
  // it returns projects where the user has projectAdmin OR projectMember access.
  // We must filter LOCALLY on accessLevels.projectAdmin === true.
  const all = [];
  let offset = 0;
  while (true) {
    const url = `${ADMIN_V1}/accounts/${accountId}/users/${LUIS_ACC_USER_ID}/projects?limit=200&offset=${offset}`;
    const r = await fetch(url, { headers: { Authorization: `Bearer ${token2Leg}` } });
    const data = await r.json();
    if (!data.results) throw new Error(`Admin v1 user-projects failed: ${r.status} ${JSON.stringify(data).slice(0, 200)}`);
    all.push(...data.results);
    const total = data.pagination?.totalResults ?? 0;
    offset += data.pagination?.limit ?? 200;
    if (offset >= total) break;
  }
  const actuallyAdmin = all.filter((p) => p.accessLevels?.projectAdmin === true);
  log(`Listed ${all.length} projects (member-or-admin); ${actuallyAdmin.length} with actual projectAdmin=true.`);
  return actuallyAdmin;
}

// 2-leg token (scripts/dc-ingest-where-i-admin.cjs:107-122):
//   grant_type=client_credentials, scope='account:read data:read'
//   POST https://developer.api.autodesk.com/authentication/v2/token
//   client_id = process.env.APS_CLIENT_ID, client_secret = process.env.APS_CLIENT_SECRET
//   This is DIFFERENT from the 3-leg user token used for DC submission.
```

```typescript
export interface AdminProject {
  id: string;                    // project UUID (no "b." prefix — matches DC IDs)
  name: string;
  status: string;                // 'active' | 'archived' | etc.
  createdAt: string;             // ISO from API; orchestrator parses to Date
  accessLevels: { projectAdmin: boolean; [k: string]: boolean };
}

export interface DiscoverOptions {
  accountId: string;             // APS hub/account id (no "b." prefix)
  userId: string;                // LUIS_ACC_USER_ID (env var)
  token2Leg: string;             // pre-fetched 2-leg bearer
  fetchImpl?: typeof fetch;      // injectable for tests; defaults to globalThis.fetch
  adminBase?: string;            // defaults to 'https://developer.api.autodesk.com/construction/admin/v1'
  pageLimit?: number;            // defaults to 200
}

export async function discoverAdminProjects(
  opts: DiscoverOptions,
): Promise<AdminProject[]>;

// Plus a 2-leg-token helper if not already exported from lib/server/aps-oauth.ts:
export async function get2LegToken(clientId: string, clientSecret: string, fetchImpl?: typeof fetch): Promise<string>;
```

```typescript
const accountId = process.env.APS_HUB_ID || process.env.ACC_ACCOUNT_ID;
const userId = process.env.LUIS_ACC_USER_ID;
if (!accountId || !userId) {

// Upsert AccDcBackfillProgress for any admin project not yet tracked.
const knownIds = new Set(existing.map((p) => p.projectId));

      latestCovered: null,

  });
}
```
